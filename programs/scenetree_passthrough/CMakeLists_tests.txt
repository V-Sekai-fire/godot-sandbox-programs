cmake_minimum_required(VERSION 3.15)
project(scenetree_passthrough_tests)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find required packages
find_package(CURL REQUIRED)

# Test executables
add_executable(test_scenetree_rpc test/test_scenetree_rpc.cpp)
target_link_libraries(test_scenetree_rpc PRIVATE CURL::libcurl)

# RapidCheck tests (if available)
find_package(rapidcheck QUIET)
if(rapidcheck_FOUND)
    add_executable(test_jsonrpc_compliance test/test_jsonrpc_compliance.cpp)
    target_link_libraries(test_jsonrpc_compliance PRIVATE rapidcheck)
endif()

# Build targets
add_custom_target(tests
    DEPENDS test_scenetree_rpc
)

if(rapidcheck_FOUND)
    add_dependencies(tests test_jsonrpc_compliance)
endif()

# Install targets
install(TARGETS test_scenetree_rpc DESTINATION bin)
if(rapidcheck_FOUND)
    install(TARGETS test_jsonrpc_compliance DESTINATION bin)
endif()
